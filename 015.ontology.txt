            <script_powershell><![CDATA[# =============================================================================
# TIER 0 - Auto-install PowerShell: Ollama + Claude Code + CCR
# Compatible: Windows 10+, PowerShell 5.1+
# Usage: .\install_tier0.ps1 -OpenRouterKey "sk-or-YOUR_KEY"
# =============================================================================

param(
    [string]$OpenRouterKey = "sk-or-REPLACE_WITH_YOUR_KEY",
    [switch]$SmallModel = $false
)

$ErrorActionPreference = "Stop"

function Write-Step($num, $total, $msg) {
    Write-Host "`n[$num/$total] $msg" -ForegroundColor Cyan
}

function Test-Command($cmd) {
    return (Get-Command $cmd -ErrorAction SilentlyContinue) -ne $null
}

Write-Host "=" * 60 -ForegroundColor Green
Write-Host " INSTALL TIER 0: Ollama + Claude Code + Router" -ForegroundColor Green
Write-Host "=" * 60 -ForegroundColor Green

# --- 1. Node.js ---
Write-Step 1 7 "Check/Install Node.js..."
if (-not (Test-Command "node")) {
    Write-Host "Node.js missing. Installing via winget..." -ForegroundColor Yellow
    winget install --id OpenJS.NodeJS.LTS -e --silent
    $env:Path = [System.Environment]::GetEnvironmentVariable("Path","Machine") + ";" + [System.Environment]::GetEnvironmentVariable("Path","User")
} else {
    Write-Host "Node.js: $(node --version) - OK" -ForegroundColor Green
}

# --- 2. Claude Code ---
Write-Step 2 7 "Installing Claude Code CLI..."
npm install -g @anthropic-ai/claude-code
Write-Host "Claude Code: OK" -ForegroundColor Green

# --- 3. Ollama ---
Write-Step 3 7 "Check/Install Ollama..."
if (-not (Test-Command "ollama")) {
    Write-Host "Ollama missing. Downloading..." -ForegroundColor Yellow
    $installerPath = "$env:TEMP\OllamaSetup.exe"
    Invoke-WebRequest -Uri "https://ollama.com/download/OllamaSetup.exe" -OutFile $installerPath
    Start-Process -FilePath $installerPath -ArgumentList "/S" -Wait
    Start-Sleep -Seconds 5
    $env:Path = [System.Environment]::GetEnvironmentVariable("Path","Machine") + ";" + $env:Path
} else {
    Write-Host "Ollama: $(ollama --version) - OK" -ForegroundColor Green
}

# --- 4. Model ---
Write-Step 4 7 "Downloading local model..."
$modelName = if ($SmallModel) { "qwen3-coder:4b" } else { "qwen3-coder" }
Write-Host "Downloading $modelName (may take 10-30 minutes)..." -ForegroundColor Yellow
ollama pull $modelName
Write-Host "Model $modelName: OK" -ForegroundColor Green

# --- 5. CCR ---
Write-Step 5 7 "Installing Claude Code Router (ccr)..."
npm install -g @musistudio/claude-code-router
Write-Host "CCR: OK" -ForegroundColor Green

# --- 6. Env vars ---
Write-Step 6 7 "Setting environment variables (user level)..."
[System.Environment]::SetEnvironmentVariable("ANTHROPIC_BASE_URL", "http://localhost:11434/v1", "User")
[System.Environment]::SetEnvironmentVariable("ANTHROPIC_API_KEY", "ollama", "User")
$env:ANTHROPIC_BASE_URL = "http://localhost:11434/v1"
$env:ANTHROPIC_API_KEY = "ollama"
Write-Host "Variables set: OK" -ForegroundColor Green

# --- 7. Router config ---
Write-Step 7 7 "Creating router config..."
$configDir = "$env:APPDATA\claude-code-router"
if (-not (Test-Path $configDir)) { New-Item -ItemType Directory -Path $configDir -Force | Out-Null }

$config = @{
    providers = @{
        ollama = @{ baseUrl = "http://localhost:11434/api"; type = "ollama" }
        openrouter = @{ apiKey = $OpenRouterKey; baseUrl = "https://openrouter.ai/api/v1" }
    }
    Router = @{
        "default" = "openrouter,meta-llama/llama-3.3-70b-instruct:free"
        "private" = "ollama,qwen3-coder"
        "background" = "openrouter,deepseek/deepseek-chat:free"
    }
} | ConvertTo-Json -Depth 5

Set-Content -Path "$configDir\config.json" -Value $config -Encoding UTF8
Write-Host "Config created: OK" -ForegroundColor Green

Write-Host "`n" + "=" * 60 -ForegroundColor Green
Write-Host " INSTALLATION COMPLETE!" -ForegroundColor Green
Write-Host "=" * 60 -ForegroundColor Green
Write-Host "`n  claude --model qwen3-coder   (Ollama local)"
Write-Host "  ccr code                      (Auto router)"
Write-Host "`n  NOTE: Reopen terminal for environment variables.`n"]]></script_powershell>
